//go:build windows

package iface

import (
	"errors"
	"net"
	"os"
	"runtime"
	"syscall"
	"unsafe"

	"golang.org/x/sys/windows"
)

var fcxAdapterAddresses = fcxDefaultAdapterAddresses

func fcxDefaultAdapterAddresses() ([]*windows.IpAdapterAddresses, []byte, error) {
	var buffer []byte
	size := uint32(15000)
	for {
		buffer = make([]byte, size)
		const flags = windows.GAA_FLAG_INCLUDE_PREFIX | windows.GAA_FLAG_INCLUDE_GATEWAYS
		err := windows.GetAdaptersAddresses(
			syscall.AF_UNSPEC,
			flags,
			0,
			(*windows.IpAdapterAddresses)(unsafe.Pointer(&buffer[0])),
			&size,
		)
		if err == nil {
			if size == 0 {
				return nil, nil, nil
			}
			break
		}
		if !errors.Is(err, syscall.ERROR_BUFFER_OVERFLOW) {
			return nil, nil, os.NewSyscallError("getadaptersaddresses", err)
		}
		if size <= uint32(len(buffer)) {
			return nil, nil, os.NewSyscallError("getadaptersaddresses", err)
		}
	}
	var adapters []*windows.IpAdapterAddresses
	for current := (*windows.IpAdapterAddresses)(unsafe.Pointer(&buffer[0])); current != nil; current = current.Next {
		adapters = append(adapters, current)
	}
	return adapters, buffer, nil
}

func fcxAdapterIndex(aa *windows.IpAdapterAddresses) int {
	index := aa.IfIndex
	if index == 0 {
		index = aa.Ipv6IfIndex
	}
	return int(index)
}

func fcxAdapterAddrs(adapters []*windows.IpAdapterAddresses, targetIndex int) ([]net.Addr, error) {
	var addrs []net.Addr
	for _, aa := range adapters {
		if fcxAdapterIndex(aa) != targetIndex {
			continue
		}
		for puni := aa.FirstUnicastAddress; puni != nil; puni = puni.Next {
			if puni.Address.Sockaddr == nil || puni.DadState != windows.IpDadStatePreferred {
				continue
			}
			sa, err := puni.Address.Sockaddr.Sockaddr()
			if err != nil {
				return nil, os.NewSyscallError("sockaddr", err)
			}
			switch sa := sa.(type) {
			case *syscall.SockaddrInet4:
				addrs = append(addrs, &net.IPNet{
					IP:   net.IPv4(sa.Addr[0], sa.Addr[1], sa.Addr[2], sa.Addr[3]),
					Mask: net.CIDRMask(int(puni.OnLinkPrefixLength), 8*net.IPv4len),
				})
			case *syscall.SockaddrInet6:
				ip := make(net.IP, net.IPv6len)
				copy(ip, sa.Addr[:])
				addrs = append(addrs, &net.IPNet{
					IP:   ip,
					Mask: net.CIDRMask(int(puni.OnLinkPrefixLength), 8*net.IPv6len),
				})
			}
		}
		for pany := aa.FirstAnycastAddress; pany != nil; pany = pany.Next {
			if pany.Address.Sockaddr == nil {
				continue
			}
			sa, err := pany.Address.Sockaddr.Sockaddr()
			if err != nil {
				return nil, os.NewSyscallError("sockaddr", err)
			}
			switch sa := sa.(type) {
			case *syscall.SockaddrInet4:
				addrs = append(addrs, &net.IPAddr{
					IP: net.IPv4(sa.Addr[0], sa.Addr[1], sa.Addr[2], sa.Addr[3]),
				})
			case *syscall.SockaddrInet6:
				ip := make(net.IP, net.IPv6len)
				copy(ip, sa.Addr[:])
				addrs = append(addrs, &net.IPAddr{IP: ip})
			}
		}
	}
	return addrs, nil
}

func fcxLoadPlatformInterfaces() ([]fcxRawInterface, error) {
	adapters, backing, err := fcxAdapterAddresses()
	if err != nil {
		return nil, err
	}
	defer runtime.KeepAlive(backing)

	entries := make([]fcxRawInterface, 0, len(adapters))
	for _, aa := range adapters {
		index := fcxAdapterIndex(aa)
		ifi := net.Interface{
			Index: index,
			Name:  windows.UTF16PtrToString(aa.FriendlyName),
		}
		if aa.OperStatus == windows.IfOperStatusUp {
			ifi.Flags |= net.FlagUp | net.FlagRunning
		}
		switch aa.IfType {
		case windows.IF_TYPE_ETHERNET_CSMACD, windows.IF_TYPE_ISO88025_TOKENRING, windows.IF_TYPE_IEEE80211, windows.IF_TYPE_IEEE1394:
			ifi.Flags |= net.FlagBroadcast | net.FlagMulticast
		case windows.IF_TYPE_PPP, windows.IF_TYPE_TUNNEL:
			ifi.Flags |= net.FlagPointToPoint | net.FlagMulticast
		case windows.IF_TYPE_SOFTWARE_LOOPBACK:
			ifi.Flags |= net.FlagLoopback | net.FlagMulticast
		case windows.IF_TYPE_ATM:
			ifi.Flags |= net.FlagBroadcast | net.FlagPointToPoint | net.FlagMulticast
		}
		if aa.Mtu == 0xffffffff {
			ifi.MTU = -1
		} else {
			ifi.MTU = int(aa.Mtu)
		}
		if aa.PhysicalAddressLength > 0 && int(aa.PhysicalAddressLength) <= len(aa.PhysicalAddress) {
			ifi.HardwareAddr = make(net.HardwareAddr, aa.PhysicalAddressLength)
			copy(ifi.HardwareAddr, aa.PhysicalAddress[:aa.PhysicalAddressLength])
		}
		addrs, err := fcxAdapterAddrs(adapters, index)
		if err != nil {
			continue
		}
		entries = append(entries, fcxRawInterface{
			iface: ifi,
			addrs: addrs,
		})
	}
	return entries, nil
}
