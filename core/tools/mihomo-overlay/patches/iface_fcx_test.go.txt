package iface

import (
	"net"
	"net/netip"
	"testing"
)

func TestFCXIfaceSinglePassCachePopulatesNamesAddressesAndRoutes(t *testing.T) {
	previous := fcxLoadInterfaces
	t.Cleanup(func() {
		fcxLoadInterfaces = previous
		FlushCache()
	})

	calls := 0
	fcxLoadInterfaces = func() ([]fcxRawInterface, error) {
		calls++
		return []fcxRawInterface{
			{
				iface: net.Interface{
					Index: 5,
					MTU:   1500,
					Name:  "eth0",
					Flags: net.FlagUp | net.FlagBroadcast,
				},
				addrs: []net.Addr{
					&net.IPNet{
						IP:   net.IPv4(192, 168, 50, 10),
						Mask: net.CIDRMask(24, 32),
					},
					&net.IPNet{
						IP:   net.ParseIP("fe80::1"),
						Mask: net.CIDRMask(64, 128),
					},
					&net.IPNet{
						IP:   net.ParseIP("2001:db8::10"),
						Mask: net.CIDRMask(64, 128),
					},
					&net.IPAddr{
						IP: net.IPv4(192, 168, 50, 254),
					},
				},
			},
			{
				iface: net.Interface{
					Index: 9,
					MTU:   1420,
					Name:  "down0",
					Flags: net.FlagPointToPoint,
				},
				addrs: []net.Addr{
					&net.IPNet{
						IP:   net.IPv4(10, 99, 0, 2),
						Mask: net.CIDRMask(24, 32),
					},
				},
			},
		}, nil
	}
	FlushCache()

	ifaces, err := Interfaces()
	if err != nil {
		t.Fatal(err)
	}
	if len(ifaces) != 2 {
		t.Fatalf("expected 2 interfaces in map, got %d", len(ifaces))
	}
	if calls != 1 {
		t.Fatalf("expected single platform load, got %d", calls)
	}

	resolved, err := ResolveInterface("eth0")
	if err != nil || resolved.Index != 5 || len(resolved.Addresses) != 4 {
		t.Fatalf("unexpected eth0 interface: %+v err=%v", resolved, err)
	}
	if calls != 1 {
		t.Fatalf("cached lookup triggered extra load: %d", calls)
	}

	localOk, err := IsLocalIp(netip.MustParseAddr("192.168.50.10"))
	if err != nil || !localOk {
		t.Fatalf("expected 192.168.50.10 to be local: ok=%v err=%v", localOk, err)
	}
	anycastLocal, err := IsLocalIp(netip.MustParseAddr("192.168.50.254"))
	if err != nil || !anycastLocal {
		t.Fatalf("expected anycast 192.168.50.254 to be local: ok=%v err=%v", anycastLocal, err)
	}
	downLocal, err := IsLocalIp(netip.MustParseAddr("10.99.0.2"))
	if err != nil || downLocal {
		t.Fatalf("down interface address must not enter active addr table: ok=%v err=%v", downLocal, err)
	}

	bySubnet, err := ResolveInterfaceByAddr(netip.MustParseAddr("192.168.50.99"))
	if err != nil || bySubnet.Name != "eth0" {
		t.Fatalf("subnet LPM failed: %+v err=%v", bySubnet, err)
	}
	if _, err := ResolveInterfaceByAddr(netip.MustParseAddr("10.99.0.99")); err == nil {
		t.Fatal("down interface prefix should not match ResolveInterfaceByAddr")
	}

	v4, err := resolved.PickIPv4Addr(netip.Addr{})
	if err != nil || v4.Addr() != netip.MustParseAddr("192.168.50.10") {
		t.Fatalf("unexpected IPv4 pick: %v err=%v", v4, err)
	}
	v6, err := resolved.PickIPv6Addr(netip.Addr{})
	if err != nil || v6.Addr() != netip.MustParseAddr("2001:db8::10") {
		t.Fatalf("expected non-link-local IPv6 fallback, got %v err=%v", v6, err)
	}

	FlushCache()
	if _, err := Interfaces(); err != nil {
		t.Fatal(err)
	}
	if calls != 2 {
		t.Fatalf("expected FlushCache to trigger exactly 1 reload, got %d total calls", calls)
	}
}
