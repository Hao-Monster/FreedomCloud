//go:build windows

package listener

import (
	"errors"
	"net"
	"strings"
	"unsafe"

	"golang.org/x/sys/windows"
)

var (
	fcxIphlpapiDLL                   = windows.NewLazySystemDLL("iphlpapi.dll")
	fcxProcConvertInterfaceAliasLuid = fcxIphlpapiDLL.NewProc("ConvertInterfaceAliasToLuid")
)

func fcxIsMissingInterfaceErr(err error) bool {
	return errors.Is(err, windows.ERROR_FILE_NOT_FOUND) ||
		errors.Is(err, windows.ERROR_INVALID_PARAMETER) ||
		errors.Is(err, windows.ERROR_NOT_FOUND)
}

func fcxPlatformTunInterfaceUsable(device string) bool {
	if device == "" {
		return true
	}
	aliasPtr, err := windows.UTF16PtrFromString(device)
	if err == nil {
		var luid uint64
		ret, _, _ := fcxProcConvertInterfaceAliasLuid.Call(
			uintptr(unsafe.Pointer(aliasPtr)),
			uintptr(unsafe.Pointer(&luid)),
		)
		if ret == 0 {
			row := windows.MibIfRow2{InterfaceLuid: luid}
			if err := windows.GetIfEntry2Ex(windows.MibIfEntryNormalWithoutStatistics, &row); err == nil {
				return row.OperStatus == windows.IfOperStatusUp
			} else if fcxIsMissingInterfaceErr(err) {
				return false
			}
		} else if fcxIsMissingInterfaceErr(windows.Errno(ret)) {
			return false
		}
	}
	interfaces, err := net.Interfaces()
	if err != nil {
		return true
	}
	for _, adapter := range interfaces {
		if strings.EqualFold(adapter.Name, device) {
			return adapter.Flags&net.FlagUp != 0
		}
	}
	return false
}
