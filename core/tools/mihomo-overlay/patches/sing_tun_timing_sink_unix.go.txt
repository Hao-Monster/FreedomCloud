//go:build unix

package sing_tun

import (
	"golang.org/x/sys/unix"
	"os"
)

// Use a non-standard descriptor so EPIPE on a lost diagnostic reader cannot
// trigger Go's special SIGPIPE handling for writes to stdout/stderr.
func fcxTimingSink() (*os.File, error) {
	fd, err := unix.FcntlInt(os.Stderr.Fd(), unix.F_DUPFD_CLOEXEC, 3)
	if err != nil {
		return nil, err
	}
	return os.NewFile(uintptr(fd), "fcx-timing"), nil
}
