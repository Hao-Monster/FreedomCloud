func TestFCXTimingPersistsAtWarningLogLevel(t *testing.T) {
	if os.Getenv("FCX_TEST_TIMING_STDERR") == "1" {
		log.SetLevel(log.WARNING)
		FCXEmitStartupTiming("test.stage", time.Millisecond, "ok", "", nil)
		// The parent acknowledges after it sees the flushed timing line.
		var ack [1]byte
		if _, err := os.Stdin.Read(ack[:]); err != nil {
			os.Exit(2)
		}
		os.Exit(0)
	}
	ctx, cancel := context.WithTimeout(context.Background(), 15*time.Second)
	defer cancel()
	cmd := exec.CommandContext(ctx, os.Args[0], "-test.run=^TestFCXTimingPersistsAtWarningLogLevel$")
	cmd.Env = append(os.Environ(), "FCX_TEST_TIMING_STDERR=1")
	input, err := cmd.StdinPipe()
	if err != nil {
		t.Fatal(err)
	}
	stderr, err := cmd.StderrPipe()
	if err != nil {
		t.Fatal(err)
	}
	if err = cmd.Start(); err != nil {
		t.Fatal(err)
	}
	line, readErr := bufio.NewReader(stderr).ReadString('\n')
	input.Write([]byte("x"))
	input.Close()
	if err = cmd.Wait(); err != nil {
		t.Fatal(err)
	}
	if readErr != nil || !strings.Contains(line, "stage=test.stage") {
		t.Fatalf("timing stderr: %q %v", line, readErr)
	}
}

func TestFCXDeviceTimingHookForwardsSubStagesAndSchema2Fields(t *testing.T) {
	if tun.FCXDeviceTimingHook == nil {
		t.Fatal("expected tun.FCXDeviceTimingHook to be wired by sing_tun init")
	}
	t.Setenv("FCX_STARTUP_ID", "corr_ab12-CD34")
	type record struct {
		stage   string
		outcome string
	}
	var got []record
	var lines []string
	previous := fcxSingTunTimingHook
	fcxSingTunTimingHook = func(stage string, _ time.Duration, outcome string) {
		got = append(got, record{stage: stage, outcome: outcome})
	}
	restoreLineHook := SetFCXStartupTimingLineHookForTest(func(line string) {
		lines = append(lines, line)
	})
	defer func() {
		fcxSingTunTimingHook = previous
		restoreLineHook()
	}()

	started := time.Now().Add(-15 * time.Millisecond)
	tun.FCXDeviceTimingHook("sing_tun.device.wintun_create", started, nil)
	tun.FCXDeviceTimingHook("sing_tun.device.ip_dns_config", started, fmt.Errorf("dns fail: %w", syscall.Errno(5)))

	if len(got) != 2 || len(lines) != 2 {
		t.Fatalf("expected 2 timing events and lines, got records=%d lines=%d", len(got), len(lines))
	}
	if got[0].stage != "sing_tun.device.wintun_create" || got[0].outcome != "ok" {
		t.Fatalf("unexpected first record: %+v", got[0])
	}
	if !strings.Contains(lines[0], "schema=2 event=stage_end correlationId=corr_ab12-CD34 ") ||
		strings.Contains(lines[0], "error=") {
		t.Fatalf("unexpected ok schema=2 line: %s", lines[0])
	}
	if got[1].stage != "sing_tun.device.ip_dns_config" || got[1].outcome != "error" {
		t.Fatalf("unexpected second record: %+v", got[1])
	}
	if !strings.Contains(lines[1], `outcome=error errorType=`) ||
		!strings.Contains(lines[1], `dns fail:`) {
		t.Fatalf("missing structured error fields on failure line: %s", lines[1])
	}
	if runtime.GOOS == "windows" && !strings.Contains(lines[1], "win32Error=5") {
		t.Fatalf("expected win32Error=5 on windows, got: %s", lines[1])
	}
}

func TestFCXSanitizeCorrelationIDRejectsUnsafeValues(t *testing.T) {
	if got := tun.FCXSanitizeCorrelationID(""); got != "unknown" {
		t.Fatalf("empty correlationId should normalize to unknown, got %q", got)
	}
	if got := tun.FCXSanitizeCorrelationID("  0123456789abcdef-_AZ  "); got != "0123456789abcdef-_AZ" {
		t.Fatalf("valid correlationId rejected: %q", got)
	}
	for _, bad := range []string{
		"bad space",
		"bad\nnewline",
		"bad=kv",
		`"quoted"`,
		strings.Repeat("a", 65),
	} {
		if got := tun.FCXSanitizeCorrelationID(bad); got != "invalid" {
			t.Fatalf("unsafe correlationId %q accepted as %q", bad, got)
		}
	}
}

func TestFCXAddAutoRoutesFailsOnIntermediateErrorAndIgnoresBenignExists(t *testing.T) {
	v4Route := netip.MustParsePrefix("0.0.0.0/0")
	v6Route := netip.MustParsePrefix("::/0")
	gw4 := netip.MustParseAddr("198.18.0.2")
	gw6 := netip.MustParseAddr("fdfe:dcba:9876::2")
	benignExists := errors.New("route object already exists")
	fatalPerm := errors.New("access denied")

	// Intermediate IPv4 failure must NOT be overwritten by succeeding IPv6 route.
	var attempted []netip.Prefix
	err := tun.FCXAddAutoRoutes(
		[]netip.Prefix{v4Route, v6Route},
		gw4,
		gw6,
		func(dst netip.Prefix, nextHop netip.Addr, metric uint32) error {
			attempted = append(attempted, dst)
			if dst == v4Route {
				if nextHop != gw4 || metric != 0 {
					t.Fatalf("unexpected v4 gateway/metric: %s %d", nextHop, metric)
				}
				return fatalPerm
			}
			return nil
		},
		func(candidate error) bool { return errors.Is(candidate, benignExists) },
	)
	if !errors.Is(err, fatalPerm) || !strings.Contains(err.Error(), "0.0.0.0/0") {
		t.Fatalf("expected intermediate IPv4 route failure to propagate, got %v", err)
	}
	if len(attempted) != 1 {
		t.Fatalf("expected fail-fast on first fatal route error, attempted=%v", attempted)
	}

	// Benign duplicate-route error on intermediate prefix must continue and succeed.
	attempted = nil
	err = tun.FCXAddAutoRoutes(
		[]netip.Prefix{v4Route, v6Route},
		gw4,
		gw6,
		func(dst netip.Prefix, nextHop netip.Addr, metric uint32) error {
			attempted = append(attempted, dst)
			if dst == v4Route {
				return benignExists
			}
			if nextHop != gw6 || metric != 0 {
				t.Fatalf("unexpected v6 gateway/metric: %s %d", nextHop, metric)
			}
			return nil
		},
		func(candidate error) bool { return errors.Is(candidate, benignExists) },
	)
	if err != nil {
		t.Fatalf("expected benign duplicate route error to be ignored, got %v", err)
	}
	if len(attempted) != 2 {
		t.Fatalf("expected both IPv4 and IPv6 routes to be processed, got %v", attempted)
	}
}

func TestFCXTimingReportsPreviousOutputFailure(t *testing.T) {
	output := fcxTimingOutput{lines: make(chan string, 2)}
	output.enqueue("first")
	output.enqueue("next")
	output.enqueue("discarded")
	w := new(fcxFailOnceWriter)
	done := make(chan struct{})
	go func() {
		output.drain(w)
		close(done)
	}()
	deadline := time.Now().Add(2 * time.Second)
	for len(output.lines) > 0 && time.Now().Before(deadline) {
		time.Sleep(time.Millisecond)
	}
	output.enqueue("clean-after-recovery")
	close(output.lines)
	<-done
	got := w.String()
	if !strings.Contains(got, "next droppedTimingLines=1 priorWriteFailures=1\n") {
		t.Fatalf("missing recovery counters: %s", got)
	}
	if !strings.Contains(got, "clean-after-recovery\n") {
		t.Fatalf("subsequent clean record remained poisoned by prior counters: %s", got)
	}
}

type fcxFailOnceWriter struct {
	failed bool
	text   strings.Builder
}

func (w *fcxFailOnceWriter) Write(p []byte) (int, error) {
	if !w.failed {
		w.failed = true
		return 0, errors.New("test write failure")
	}
	return w.text.Write(p)
}
func (w *fcxFailOnceWriter) String() string { return w.text.String() }
